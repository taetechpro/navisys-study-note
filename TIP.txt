## Apply KITTI LiDAR segmentation to your own collected data

# Current run_vio expects a KITTI-like stereo/IMU/LiDAR dataset.
# Recommended converted layout:
#   data/my_dataset/my_drive_sync/
#     image_00/timestamps.txt
#     image_00/data/0000000000.png
#     image_01/timestamps.txt
#     image_01/data/0000000000.png
#     oxts/timestamps.txt
#     oxts/data/0000000000.txt
#     velodyne_points/data/0000000000.bin
#
# LiDAR .bin format:
#   little-endian float32 x, y, z, intensity
#   x forward, y left, z up, unit = meter
#
# Important:
# - Velodyne frames are matched by the same index as camera frames.
# - If your sensor uses a different axis convention, convert it before writing .bin.
# - For LiDAR-only logs, add a small LiDAR-only executable around LidarSegmenter.

# Copy an existing KITTI config and edit dataset/output/calibration:
Copy-Item D:\02_research\04_cpp_seg_msckf_vio\config\kitti_raw_2011_09_26_drive_0117_local_max_frame.yaml D:\02_research\04_cpp_seg_msckf_vio\config\my_drive.yaml

# Then edit config/my_drive.yaml:
#   dataset_type: "kitti_raw"
#   dataset: "/mnt/d/02_research/04_cpp_seg_msckf_vio/data/my_dataset/my_drive_sync"
#   output:  "/mnt/d/02_research/04_cpp_seg_msckf_vio/results/my_drive"
#   max_frames: 0
#   cam0/cam1 intrinsics and T_cam_imu: replace with your calibration

# Run your converted dataset:
wsl bash -lc 'cd /mnt/d/02_research/04_cpp_seg_msckf_vio && mkdir -p results/my_drive && ./build_verify_rerun/run_vio config/my_drive.yaml --segment-lidar --segment-every 1 --rerun-save results/my_drive/lidar_segments.rrd --rerun-image-every 10'

# Open your result:
rerun D:\02_research\04_cpp_seg_msckf_vio\results\my_drive\lidar_segments.rrd
